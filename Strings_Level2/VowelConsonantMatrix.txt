import java.util.Scanner;

class VowelConsonantMatrix {
    public static String checkCharType(char ch) {
        if (ch >= 'A' && ch <= 'Z') {
            ch = (char) (ch + 32);
        }
        if (ch >= 'a' && ch <= 'z') {
            if (ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u') {
                return "Vowel";
            }
            return "Consonant";
        }
        return "Not a Letter";
    }

    public static String[][] getAnalysisMatrix(String text) {
        String[][] matrix = new String[text.length()][2];
        for (int i = 0; i < text.length(); i++) {
            char ch = text.charAt(i);
            matrix[i][0] = String.valueOf(ch);
            matrix[i][1] = checkCharType(ch);
        }
        return matrix;
    }

    public static void displayMatrix(String[][] matrix) {
        System.out.printf("%-10s %-15s%n", "Character", "Type");
        System.out.println("--------------------------");
        for (int i = 0; i < matrix.length; i++) {
            System.out.printf("%-10s %-15s%n", matrix[i][0], matrix[i][1]);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter text: ");
        String text = sc.nextLine();

        String[][] matrix = getAnalysisMatrix(text);
        displayMatrix(matrix);
        sc.close();
    }
}
