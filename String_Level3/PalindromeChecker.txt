import java.util.Scanner;

class PalindromeChecker {
    public static boolean checkIterative(String str) {
        int start = 0;
        int end = str.length() - 1;
        while (start < end) {
            if (str.charAt(start) != str.charAt(end)) {
                return false;
            }
            start++;
            end--;
        }
        return true;
    }

    public static boolean checkRecursive(String str, int start, int end) {
        if (start >= end) {
            return true;
        }
        if (str.charAt(start) != str.charAt(end)) {
            return false;
        }
        return checkRecursive(str, start + 1, end - 1);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter text: ");
        String text = sc.next();
        
        boolean iterResult = checkIterative(text);
        boolean recResult = checkRecursive(text, 0, text.length() - 1);
        
        System.out.println("Iterative Check Result: " + iterResult);
        System.out.println("Recursive Check Result: " + recResult);
        sc.close();
    }
}
